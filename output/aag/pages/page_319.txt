آسان عربی گرامر ــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ سباق نمبر: 77 ــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ لفیف

ہوتا ہے۔ جیسے باب ضَرَبَ میں وَقَى يَقِيْ سے وَقَى يَقِيْ باب حَسِبَ میں وَلِيَ يَوْلِي سے وَلِيَ يَلِي (قریب ہونا) اور باب سَمِعَ میں وَهِيَ يَوْهَى سے وَهِيَ يَوْهَى (کمزور ہونا) ہو جائے گا۔

77.6: لَفِيف مفروق میں مثال اور ناقص دونوں کے قواعد کے اطلاق کا ایک خاص اثر یہ ہوتا ہے کہ اس کے امر حاضر کے پہلے صیغے میں فعل کا صرف عین کلمہ باقی بچتا ہے۔ مثلاً وَقَى يَقِيْ سے مضارع يَقِيْ کے بجائے يَقي استعمال ہوتا ہے۔ اس سے جب فعل امر بنانے کے لیے علامت مضارع گراتے ہیں توقِي باقی بچتا ہے۔ پھر جب لام کلمہ "ی" کو مجزوم کرتے ہیں تو وہ بھی گر جاتی ہے۔ اس طرح فعل امر "قِ" (توبچا) استعمال ہوتا ہے۔

77.7: اوپر آپ کو یہ بھی بتایا گیا ہے کہ لَفِيف مقرون وہ ہوتا ہے جس میں حروف علت باہم قرین یعنی ساتھ ساتھ ہوتے ہیں۔ عام طور پر کسی مادّے کے عین اور لام کلمہ پر حروف علت یکجا ہوتے ہیں۔ فاء اور عین کلمہ پر یکجا ہونے والے مادّے بہت کم ہیں اور جو چند ایسے مادّے ہیں بھی تو عموماً ان سے کوئی فعل استعمال نہیں ہوتا۔ مثلاً قرآن کریم میں ایسے مادّے دو لفظوں وَيْلٌ (خرابی، تباہی وغیرہ) اور يَوْمُئِذٍ (دن) میں آئے ہیں۔ اگرچہ عربی ڈکشنریوں میں ان دونوں مادّوں سے ایک آدھ فعل بھی مذكور ہوا ہے لیکن قرآن کریم میں ان سے ماخوذ کوئی صیغہ فعل کہیں وارد نہیں ہوا۔ لہٰذا عربی گرامر میں جب لَفِيف مقرون کا ذکر ہوتا ہے تو اس سے مراد وہی مادّہ ہوتا ہے جس میں عین کلمہ اور لام کلمہ حرفِ علت ہوں۔

77.8: لَفِيف مقرون میں عین کلمہ پر "وَ" اور لام کلمہ پر "ی" ہی ہوتی ہے۔ ایسا نہیں ہوتا کہ عین کلمہ پر "ی" اور لام کلمہ پر "وَ" ہو، اور یہ مجرد کے صرف دو ابواب ضَرَبَ اور سَمِعَ سے آتا ہے۔ دونوں ابواب میں عین کلمہ کا "وَ" تبدیل نہیں ہوتا جبکہ لام کلمہ کی "ی" میں قواعد کے مطابق تبدیلی آتی ہے۔ مثلاً ضَرَبَ میں غَوَى يَعْوَى سے غَوَى يَعْوِيْ سے يَعْوِيْ (بہک جانا) اور سَمِعَ میں سَوَى يَسْوَى سے سَوَى يَسْوَى، يَسْوَى (برابر ہونا) ہو جائے گا۔

77.9: بعض دفعہ لَفِيف مقرون مضاعف بھی ہوتا ہے یعنی عین کلمہ اور لام کلمہ دونوں "وَ" یادونوں "ی" ہوتے ہیں۔ مثلاً اَجْ و جس کا اسم الْجَوُّ (زمین اور آسمانوں کی درمیانی فضا) قرآن کریم میں استعمال ہوا ہے۔ اسی طرح ح ی ی اور ع ی ی بھی قرآن کریم میں آئے ہیں۔ ایسی صورت میں مثلین کا ادغام اور فک ادغام دونوں جائز ہیں، یعنی حَيِيَ يَحْيَى سے حَيِيَ يَحْيَى (زندہ ہونا/ رہنا) بھی درست ہے اور حَيَّ يَحْيَى بھی درست ہے۔ اسی طرح عَيِيَ يَعْيَى سے عَيِيَ يَعْيَى (تھک کر رہ جانا، عاجز ہونا) اور عَيَّ يَعْيُّ، يَعْيَىٰ دونوں درست ہیں۔
