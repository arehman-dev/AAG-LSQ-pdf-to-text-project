آسان عربی گرامر ــــــــــــــــــــــــــــــــــــــــ سبَق نمبر: 74 ــــــــــــــــــــــــــــــــ ناقص (حصّہ دوم "ماضی معروف")

ii) مضارع معروف میں مُثَنّیٰ کے چاروں صیغے یعنی يَفْعَلانِ اور تَفْعَلانِ کے اوزان مستثنیٰ ہیں۔ مثلاً يَدْعُوآنِ، تَدْعُوآنِ اور يَلْقَيَانِ، تَلْقَيَانِ بھی تبدیلی کے بغیر استعمال ہوں گے۔

73:6 ناقص کا پہلا قاعدہ یہ ہے کہ ناقص کے لام کلمہ کا حرف علت اور صیغہ کا حرف علت اگر یکجا ہو جائیں تو لام کلمہ کا حرف علت گر جاتا ہے۔ پھر عین کلمہ پر اگر فتحہ (ـَ) ہے تو وہ برقرار رہے گا، اگر ضمہ (ـُ) یا کسرہ (ـِ) ہے تو اسے صیغہ کے حرف علت کے مناسب رکھنا ہوگا۔ اب اس قاعدہ کو دونوں طرح کی مثالوں سے سمجھ لیں۔

73:7 پہلے وہ مثال لے لیں جس میں عین کلمہ پر فتحہ (ـَ) ہوتا ہے جو برقرار رہے گا۔ دَعَا (دَعَو) کے جمع مذکر غائب کے صیغہ میں اصلی شکل دَعَوْواً بنتی ہے۔ اس کے لام کلمہ کا "و" گرے گا تو دَعَوا باقی بچا۔ عین کلمہ کا فتحہ (ـَ) برقرار رہے گا اس لیے یہ دَعَوْا ہی استعمال ہوگا۔ اسی طرح رَمَى (رَمَي) "اس نے پھینکا" کی جمع مذکر غائب کے صیغہ میں اصلی شکل رَمَيْواً ہوگی۔ لام کلمہ کی "ی" گرے گی تو رَمَوْا باقی بچے گا اور یہ اسی طرح استعمال ہوگا۔

73:8 اب وہ مثال لے لیں جس میں عین کلمہ پر کسره (ـِ) یا ضمه (ـُ) ہوتا ہے۔ لَقِيَ (ملاقات کرنا) کی جمع مذکر غائب کے صیغہ میں اصلی شکل لَقِيُوا ہوگی۔ لام کلمہ کی "ی" گرے گی تو لَقُوا باقی بچے گا۔ عین کلمہ کا کسره (ـِ) صیغہ کے "و" سے مناسبت نہیں رکھتا اس لیے اس کو ضمہ (ـُ) میں تبدیل کرکے لَقُوا استعمال کریں گے۔ سَرُوَ (شریف ہونا) کی جمع مذکر غائب کے صیغہ میں اصلی شکل سَرُوُوا ہوگی۔ لام کلمہ کا "و" گرا تو سَرُوا باقی بچا۔ عین کلمہ کا ضمہ (ـُ) صیغہ کے "و" کے مناسب ہے اس لیے سَرُوا ہی استعمال ہوگا۔

73:9 ناقص کے ماضی معروف میں واحد مؤنث غائب کے صیغہ یعنی فَعَلَتْ کے وزن میں کبھی تبدیلی ہوتی ہے اور کبھی نہیں ہوتی۔ اس کا اصول سمجھ لیں۔ ناقص کے جن افعال میں اجوف کے قاعدہ نمبر 3 (الف) کے اطلاق کی گنجائش ہوتی ہے وہاں تبدیلی ہوتی ہے لیکن قاعدہ نمبر 3 (ب) کا اطلاق نہیں ہوتا۔ جن افعال میں اس قاعدہ کے اطلاق کی گنجائش نہیں ہوتی وہ تبدیلی کے بغیر استعمال ہوتے ہیں۔ اس اصول کو مثالوں کی مدد سے سمجھ لیں۔

73:10 دَعَا (دَعَو) کے واحد مؤنث غائب کے صیغہ میں اصلی شکل دَعوَتْ ہوگی۔ حرف علت متحرک اور ماقبل فتحہ (ـَ) ہونے کی وجہ سے "و" الف میں تبدیل ہو گا لیکن ما بعد ساکن ہونے کی وجہ سے گر جائے گا۔ اس طرح یہ دَعَتْ استعمال ہوگا۔ لَقِيَ کی صیغہ مذکور میں اصلی شکل لَقِيَتْ ہوگی۔ یہاں حرف علت متحرک کے ماقبل حرف پر کسره (ـِ) ہے اس لیے قاعدہ کا اطلاق نہیں ہوگا اور لَقِيَتْ ہی استعمال ہوگا۔
